// FIRESTORE SECURITY RULES
// Go to: Firebase Console → Your Project → Firestore → Rules
// Replace existing rules with these:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Active trips — only the rider and assigned driver can read/write
    match /active_trips/{tripId} {
      allow read, write: if request.auth != null;
    }

    // Ride requests — all authenticated users (drivers) can read pending ones
    // Only authenticated users can create/update
    match /ride_requests/{requestId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null;
    }

    // Driver locations — authenticated drivers can write their own location
    // Authenticated riders can read to see nearby drivers
    match /driver_locations/{driverId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
  }
}

// NOTE: request.auth != null is satisfied by Firebase Anonymous Auth
// which the app now signs in automatically on startup.
// This is safe — the real user identity is managed by Laravel Sanctum.
